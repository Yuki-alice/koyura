```kotlin
// shared/design-system/src/commonMain/kotlin/theme/KmTheme.kt
package app.koyura.designsystem.theme

import androidx.compose.material3.ColorScheme
import androidx.compose.material3.MaterialExpressiveTheme
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.Typography
import androidx.compose.material3.darkColorScheme
import androidx.compose.material3.lightColorScheme
import androidx.compose.runtime.Composable
import androidx.compose.ui.graphics.Color
import com.materialkolor.DynamicMaterialExpressiveTheme
import com.materialkolor.PaletteStyle

/** 对应 Komikku 的 AppTheme 枚举（19 个有名字的主题 + CUSTOM + MONET）。 */
sealed interface KmAppTheme {
    /** 默认动态取色（Komikku 的默认值，不是 DEFAULT 静态主题） */
    data object Monet : KmAppTheme

    /** 自定义种子色 + PaletteStyle */
    data class Custom(val seed: Color, val style: PaletteStyle = PaletteStyle.Fidelity) : KmAppTheme

    /** 静态主题 */
    data class Static(val scheme: KmColorScheme) : KmAppTheme
}

/** 一套静态主题的亮/暗两套 scheme。 */
data class KmColorScheme(
    val id: String,
    val dark: ColorScheme,
    val light: ColorScheme,
    /** 显式定义的角色数（用于诊断"半残主题"） */
    val definedRoles: Int,
)

@Composable
fun KmTheme(
    appTheme: KmAppTheme = KmAppTheme.Monet,
    darkTheme: Boolean = isSystemInDarkTheme(),
    amoled: Boolean = false,
    /** 封面取色：非 null 时启用局部动态主题 */
    seedColor: Color? = null,
    coverBasedStyle: PaletteStyle = PaletteStyle.Vibrant,
    typography: Typography = MaterialTheme.typography,
    content: @Composable () -> Unit,
) {
    if (seedColor != null) {
        // Komikku: TachiyomiTheme(seedColor != null) → DynamicMaterialExpressiveTheme
        DynamicMaterialExpressiveTheme(
            seedColor = seedColor,
            isDark = darkTheme,
            isAmoled = amoled,
            style = coverBasedStyle,
            typography = typography,
            animate = true,
            content = content,
        )
        return
    }

    val scheme = when (appTheme) {
        KmAppTheme.Monet -> null   // 交给 materialkolor 从壁纸/系统取
        is KmAppTheme.Custom -> null
        is KmAppTheme.Static -> (if (darkTheme) appTheme.scheme.dark else appTheme.scheme.light)
            .applyAmoled(
                // Komikku: overrideDarkSurfaceContainers = appTheme != MONET
                overrideContainers = amoled,
            )
    }

    when (appTheme) {
        KmAppTheme.Monet -> DynamicMaterialExpressiveTheme(
            // KMP 没有 WallpaperManager；桌面/iOS 需要一个默认种子
            seedColor = KmDefaultMonetSeed,
            isDark = darkTheme,
            isAmoled = amoled,
            style = PaletteStyle.TonalSpot,
            typography = typography,
            content = content,
        )

        is KmAppTheme.Custom -> DynamicMaterialExpressiveTheme(
            seedColor = appTheme.seed,
            isDark = darkTheme,
            isAmoled = amoled,
            style = appTheme.style,
            typography = typography,
            content = content,
        )

        is KmAppTheme.Static -> MaterialExpressiveTheme(
            colorScheme = scheme!!,
            typography = typography,
            content = content,
        )
    }
}

private val KmDefaultMonetSeed = Color(0xFF2979FF)   // Komikku Default 主题的 key colour
```

### AMOLED

```kotlin
// shared/design-system/src/commonMain/kotlin/theme/Amoled.kt
fun ColorScheme.applyAmoled(overrideContainers: Boolean): ColorScheme {
    val base = copy(
        background = Color.Black,
        onBackground = Color.White,
        surface = Color.Black,
        onSurface = Color.White,
    )
    if (!overrideContainers) return base
    return base.copy(
        surfaceVariant = Color(0xFF0C0C0C),
        surfaceContainerLowest = Color(0xFF0C0C0C),
        surfaceContainerLow = Color(0xFF0C0C0C),
        surfaceContainer = Color(0xFF0C0C0C),
        surfaceContainerHigh = Color(0xFF131313),
        surfaceContainerHighest = Color(0xFF1B1B1B),
    )
}
```

> 注意：`applyAmoled` 只在**暗色**下有意义。亮色主题开 AMOLED 应当无效（Komikku 的 `getColorScheme` 里 `if (!isDark) return lightScheme` 在最前面）。

    for v in ("dark", "light"):

        print(f"    {v} = {fn}(")

## 9.3 CMP 的断点

### 问题

`isTabletUi()` 依赖 `Configuration.smallestScreenWidthDp`（Android 独有）。CMP 的 Desktop/iOS 没有。

### 方案 A（推荐）：`BoxWithConstraints` + 自定义 CompositionLocal

```kotlin
// shared/design-system/src/commonMain/kotlin/layout/SmallestWidth.kt
package app.koyura.designsystem.layout

import androidx.compose.foundation.layout.BoxWithConstraints
import androidx.compose.runtime.Composable
import androidx.compose.runtime.CompositionLocalProvider
import androidx.compose.runtime.ProvidableCompositionLocal
import androidx.compose.runtime.staticCompositionLocalOf
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.Dp
import androidx.compose.ui.unit.dp

val LocalSmallestWidth: ProvidableCompositionLocal<Dp> = staticCompositionLocalOf { 360.dp }

/**
 * 在应用根调用一次。对应 Komikku DisplayExtensions.kt:21-23 的
 * `smallestScreenWidthDp >= 720`。
 */
@Composable
fun ProvideSmallestWidth(
    modifier: Modifier = Modifier,
    content: @Composable () -> Unit,
) {
    BoxWithConstraints(modifier = modifier) {
        val smallest = minOf(maxWidth, maxHeight)
        CompositionLocalProvider(
            LocalSmallestWidth provides smallest,
            content = content,
        )
    }
}

@Composable
fun isTabletUi(): Boolean = LocalSmallestWidth.current >= 720.dp
```

**注意**：`BoxWithConstraints` 会在测量阶段触发子内容重组。只能在应用根用一次，不要放进 `LazyColumn` 的 item 里。

### 方案 B：平台 `expect/actual`

```kotlin
// commonMain
expect fun platformSmallestWidthDp(): Dp

// androidMain
actual fun platformSmallestWidthDp(): Dp =
    LocalConfiguration.current.smallestScreenWidthDp.dp

// iosMain：取 min(UIScreen.main.bounds)
// jvmMain：取 window 尺寸
```

方案 B 更精确（Android 与原生完全一致），但需要三份实现。

### 桌面窗口的特殊处理

`min(w, h)` 在 21:9 超宽窗口下会判定为"手机"。建议桌面：

```kotlin
@Composable
fun isTabletUiDesktopAware(): Boolean {
    val w = LocalSmallestWidth.current
    return if (isDesktopPlatform()) {
        // 桌面用宽度判定，且门槛更高（桌面最小宽 840dp 才上双栏）
        LocalWindowWidth.current >= 840.dp
    } else {
        w >= 720.dp
    }
}
```

`⚠️缺口` Komikku 没有桌面版，这个门槛是 `project x` 自己的决定。

### 700 / 600 的自动阈值

Komikku 的 `prepareTabletUiContext()` 用 700（竖屏）/ 600（横屏）决定**是否强制覆盖** `smallestScreenWidthDp`。

CMP 里可以直接实现"用户偏好"：

```kotlin
enum class TabletUiMode { AUTOMATIC, ALWAYS, LANDSCAPE, NEVER }

@Composable
fun shouldUseTabletLayout(mode: TabletUiMode): Boolean {
    val smallest = LocalSmallestWidth.current
    val landscape = LocalWindowWidth.current > LocalWindowHeight.current
    return when (mode) {
        TabletUiMode.AUTOMATIC -> smallest >= (if (landscape) 600.dp else 700.dp)
        TabletUiMode.ALWAYS -> true
        TabletUiMode.LANDSCAPE -> landscape
        TabletUiMode.NEVER -> false
    }
}
```

> ⚠️ 注意与 Komikku 的**语义差异**：Komikku 的 `AUTOMATIC` 判定用于"是否覆盖 configuration"，最终布局判定仍是 720。上面的实现把 700/600 直接当布局断点，**在 700–719dp 区间会产生与 Komikku 不同的结果**。
>
> 要完全一致，需要两段式：先用 700/600 算 `expectedTablet`，再把 `smallest` 夹到 `[720, ∞)` 或 `[0, 719]`，最后用 720 判定。

### 配方 2 · 双栏

```kotlin
@Composable
fun KmTwoPanelBox(
    startContent: @Composable BoxScope.() -> Unit,
    endContent: @Composable BoxScope.() -> Unit,
    modifier: Modifier = Modifier,
    contentWindowInsets: PaddingValues = PaddingValues(0.dp),
) {
    BoxWithConstraints(modifier.fillMaxSize()) {
        val startPad = contentWindowInsets.calculateStartPadding(LocalLayoutDirection.current)
        val endPad = contentWindowInsets.calculateEndPadding(LocalLayoutDirection.current)
        val width = maxWidth - startPad - endPad
        val firstWidth = (width / 2).coerceAtMost(450.dp)      // ← 唯一规则
        val secondWidth = width - firstWidth

        Box(Modifier.align(Alignment.TopStart).width(firstWidth + startPad), content = startContent)
        Box(Modifier.align(Alignment.TopEnd).width(secondWidth + endPad), content = endContent)
    }
}
```

### 配方 3 · 自适应弹层

@Composable
fun KmAdaptiveSheet(
    onDismissRequest: () -> Unit,
    tablet: Boolean = isTabletUi(),
    content: @Composable () -> Unit,
) {
    Dialog(onDismissRequest = onDismissRequest, properties = DialogProperties(usePlatformDefaultWidth = false)) {
        if (tablet) {
            // 460dp 居中卡片 + alpha 350ms
            var target by remember { mutableFloatStateOf(0f) }
            val alpha by animateFloatAsState(target, tween(350))
            LaunchedEffect(Unit) { target = 1f }
            Box(
                Modifier.fillMaxSize().alpha(alpha)
                    .clickable(interactionSource = null, indication = null, onClick = onDismissRequest),
                contentAlignment = Alignment.Center,
            ) {
                Surface(
                    Modifier.requiredWidthIn(max = 460.dp)
                        .clickable(interactionSource = null, indication = null, onClick = {})
                        .padding(vertical = 16.dp),
                    shape = MaterialTheme.shapes.extraLarge,   // 28dp
                    color = MaterialTheme.colorScheme.surfaceContainerHigh,
                ) { content() }
            }
        } else {
            // 底部抽屉 + anchoredDraggable 350ms + 56dp 阈值
            // 实现见 Komikku AdaptiveSheet.kt:105-190
            KmBottomSheet(onDismissRequest, content)
        }
    }
}

### 配方 4 · 章节行（可滑动）

CMP 没有 `me.saket.swipe`。用 `anchoredDraggable` 或简单的 `SwipeToDismissBox`：

```kotlin
@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun KmChapterRow(chapter: ChapterUi, onToggleRead: () -> Unit) {
    val state = rememberSwipeToDismissBoxState(
        // 56dp 阈值 → 用 positionalThreshold
        positionalThreshold = { distance -> 56.dp.toPx().coerceAtMost(distance * 0.5f) },
        confirmValueChange = {
            if (it != SwipeToDismissBoxValue.Settled) { onToggleRead(); false } else false
        },
    )
    SwipeToDismissBox(
        state = state,
        backgroundContent = {
            Box(Modifier.fillMaxSize().background(MaterialTheme.colorScheme.primaryContainer))
        },
    ) {
        Row(
            Modifier.fillMaxWidth()
                .padding(start = 16.dp, top = 12.dp, end = 8.dp, bottom = 12.dp),
        ) {
            Column(Modifier.weight(1f), verticalArrangement = Arrangement.spacedBy(6.dp)) {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    if (!chapter.read) {
                        Box(
                            Modifier.size(8.dp).padding(end = 4.dp)
                                .background(MaterialTheme.colorScheme.primary, CircleShape),
                        )
                    }
                    Text(
                        chapter.name,
                        style = MaterialTheme.typography.bodyMedium,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis,
                        color = LocalContentColor.current.copy(
                            alpha = if (chapter.read) 0.38f else 1f,
                        ),
                    )
                }
                Text(chapter.date, style = MaterialTheme.typography.bodySmall, maxLines = 1)
            }
            KmDownloadIndicator(chapter.downloadState)
        }
    }
}
```

> ⚠️ `SwipeToDismissBox` 与 `SwipeableActionsBox` 的行为不同：前者是"滑走"，后者是"滑动执行动作后回弹"。要完全一致需要自写 `anchoredDraggable`（`AdaptiveSheet` 里有可参考的实现）。

```kotlin
@Composable
fun KmReaderPageIndicator(current: String, total: Int, modifier: Modifier = Modifier) {
    if (current.isEmpty() || total <= 0) return
    val text = "$current / $total"
    val base = TextStyle(
        color = MaterialTheme.colorScheme.primary,
        fontSize = MaterialTheme.typography.bodySmall.fontSize,
        fontWeight = FontWeight.Bold,
        letterSpacing = 1.sp,
    )
    val outline = base.copy(color = Color(45, 45, 45), drawStyle = Stroke(width = 4f))
    Box(modifier, contentAlignment = Alignment.Center) {
        Text(text, style = outline)   // 底层：描边
        Text(text, style = base)      // 上层：填充
    }
}
```

> `drawStyle = Stroke(...)` 在 CMP 1.13 的 `TextStyle` 上可用（Skia 后端）。这是跨平台一致的。

## 9.6 十个坑

### 坑 1 · 把 `MONET` 当成 `DEFAULT`

`AppTheme` 枚举第一项是 `DEFAULT`，但**默认选中是 `MONET`**（`UiPreferences.kt:20-23`）。如果照枚举顺序实现，会默认显示蓝色而不是壁纸色。

### 坑 2 · 网格列数漏了导航轨

第一版公式 `floor((w+4)/132)` 漏了 80dp 导航轨和 2×4dp 项内边距。正确：`floor((w − rail − 16 + 4) / 132)`，封面宽再减 `2×4dp`。见 [03](03-布局与响应式.md) §3.5。

### 坑 3 · 把"选中态"做成自定义药丸

Komikku 用的是 M3 `NavigationBarItem`/`NavigationRailItem` 的默认 indicator。手写药丸会做出错误的尺寸和颜色。

### 坑 4 · 混淆三套设置行

16/16（通用）、24/10（`SettingsItems`）、8+24dp（平板双栏选中行）。**按组件选，不按页面选。**

### 坑 5 · 下载进度环用 4dp 描边

必须 **13dp** + `StrokeCap.Butt` + `gapSize = 0`，否则不是"扇形"。

### 坑 6 · 详情页顶栏合并两个 alpha

标题和背景是独立的 `animateFloatAsState`，触发条件不同（`isFirstItemVisible` vs `isFirstItemVisible || isFirstItemScrolled`）。合并会丢掉"标题晚一步"的效果。

### 坑 7 · 阅读器忘了覆写 `bodyLarge`

`ReaderActivity.kt:321-323` 把 `bodyLarge` 换成 `bodySmall`。不覆写会让阅读器排版整体大 33%。

### 坑 8 · 把截图里的紫色下划线当设计

`19_settings_appearance.jpg` 和 `20_settings_library.jpg` 里的下划线是**键盘/无障碍焦点指示器**，不是设计元素。

### 坑 9 · Tab 切换用默认 fade-through

M3 默认 `materialFadeThrough` 有 0.92 → 1.0 的缩放。Komikku 显式设为 `initialScale = 1f`。不设会显得"重"。

### 坑 10 · 把 `*Fixed` 角色算进"完整主题"

M3 1.5 有 **48** 个角色，不是 36。19 套主题全都缺 12 个 `*Fixed`。如果按 36 去补，会漏掉一半。

## 设计意图（`🎯推断`）

1. **为什么先修 token 再写组件**：`DesignTokens.kt` 已经被组件消费。如果先写组件再改 token，要回头改所有引用点。
2. **为什么推荐 `BoxWithConstraints` 而非 `expect/actual`**：一份实现覆盖三端，维护成本低。精度差异只在"窗口尺寸取整"上，通常 ±1dp，视觉无感。
3. **为什么保留 19 套静态主题**：这是 Komikku 的"可玩性"来源。即使其中 4 套不完整，用户仍会觉得"有很多选择"。
4. **为什么要文档化"半残主题"**：不写清楚，实现者会以为是自己写错了，花时间排查根本不存在的 bug。
